from abc import ABC, abstractmethod
from typing import Dict, Any, List
from dataclasses import dataclass

@dataclass(frozen=True)
class ModelEvaluationResult:
    """Encapsulates immutable inference metrics with strict type annotations."""
    model_name: str
    prediction: str
    confidence: float
    status: str

class BasePredictiveModel(ABC):
    """
    Abstract Base Class establishing interface contracts for AI pipelines.
    Enforces abstraction and polymorphism across all model implementations.
    """
    def __init__(self, model_id: str, version: str) -> None:
        self._model_id: str = model_id          # Protected attribute (Encapsulation)
        self._version: str = version
        self._is_initialized: bool = False

    @property
    def model_id(self) -> str:
        """Getter demonstrating controlled access to encapsulated state."""
        return self._model_id

    @property
    def is_ready(self) -> bool:
        return self._is_initialized

    def warm_up(self) -> None:
        """Shared concrete method inherited across the class hierarchy."""
        self._is_initialized = True
        print(f"[{self._model_id} v{self._version}] Initialized and weights loaded.")

    @abstractmethod
    def predict(self, input_features: List[float]) -> ModelEvaluationResult:
        """Abstract contract that every concrete model subclass must implement."""
        pass

class CareerPathClassifier(BasePredictiveModel):
    """
    Concrete subclass demonstrating Inheritance and Polymorphic execution
    for automated career readiness classification.
    """
    def __init__(self, target_career: str, confidence_threshold: float = 0.80) -> None:
        super().__init__(model_id="CareerPathAI_Classifier", version="2.4.0")
        self.target_career: str = target_career
        self.threshold: float = confidence_threshold

    def predict(self, input_features: List[float]) -> ModelEvaluationResult:
        if not self._is_initialized:
            raise RuntimeError("Model must be initialized before inference.")

        # Polymorphic calculation specialized for career classification
        mean_score = sum(input_features) / (len(input_features) or 1)
        is_ready = mean_score >= self.threshold

        return ModelEvaluationResult(
            model_name=self.model_id,
            prediction=self.target_career if is_ready else "Prerequisite Study Required",
            confidence=round(mean_score, 4),
            status="Certified Ready" if is_ready else "Developing Competencies"
        )

class SkillGapEvaluator(BasePredictiveModel):
    """
    Second concrete subclass demonstrating Polymorphism in a shared pipeline.
    Implements identical interface with domain-specific evaluation logic.
    """
    def __init__(self, skill_name: str) -> None:
        super().__init__(model_id="SkillGap_Evaluator", version="1.1.0")
        self.skill_name: str = skill_name

    def predict(self, input_features: List[float]) -> ModelEvaluationResult:
        if not self._is_initialized:
            raise RuntimeError("Evaluator requires initialization.")

        min_score = min(input_features) if input_features else 0.0
        meets_standard = min_score >= 0.70

        return ModelEvaluationResult(
            model_name=self.model_id,
            prediction=f"Skill: {self.skill_name}",
            confidence=round(min_score, 4),
            status="Mastery Met" if meets_standard else "Gap Detected"
        )

# --- Polymorphic Inference Pipeline ---
pipeline: List[BasePredictiveModel] = [
    CareerPathClassifier(target_career="Software Engineer", confidence_threshold=0.82),
    SkillGapEvaluator(skill_name="OOP Architecture & Design")
]

for model in pipeline:
    model.warm_up()
    result = model.predict([0.88, 0.94, 0.85, 0.91])
    print(f"-> {result.model_name}: {result.prediction} | Status: {result.status} (Conf: {result.confidence})")

from typing import List, Dict, Optional

def evaluate_learner_metrics(scores: List[int], min_passing: int = 70) -> Dict[str, Any]:
    """Processes student scores using type-safe control flow and error boundaries."""
    if not scores:
        raise ValueError("Cannot compute metrics on empty score collection.")

    # Clean and filter with list comprehension
    valid_scores = [s for s in scores if 0 <= s <= 100]
    avg_score = sum(valid_scores) / len(valid_scores)
    
    # Control flow pattern matching status
    if avg_score >= 90:
        tier = "Distinction"
    elif avg_score >= min_passing:
        tier = "Competent"
    else:
        tier = "Remediation Required"

    return {
        "sample_size": len(valid_scores),
        "average": round(avg_score, 2),
        "status_tier": tier
    }

try:
    summary = evaluate_learner_metrics([85, 92, 78, 96, 88])
    print("Evaluation Summary:", summary)
except ValueError as err:
    print(f"Validation Error: {err}")

# LRU Cache Implementation with O(1) Access & Eviction
# Demonstrates: Hash Map + Doubly Linked List Architecture
# =====================================================================
class Node:
    def __init__(self, key: str, val: int):
        self.key = key
        self.val = val
        self.prev = None
        self.next = None

class LRUCache:
    def __init__(self, capacity: int):
        self.cap = capacity
        self.map = {}
        self.head = Node("", 0)
        self.tail = Node("", 0)
        self.head.next = self.tail
        self.tail.prev = self.head

    def _remove(self, node: Node) -> None:
        node.prev.next = node.next
        node.next.prev = node.prev

    def _insert(self, node: Node) -> None:
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node

    def get(self, key: str) -> int:
        if key in self.map:
            node = self.map[key]
            self._remove(node)
            self._insert(node)
            return node.val
        return -1

    def put(self, key: str, val: int) -> None:
        if key in self.map:
            self._remove(self.map[key])
        node = Node(key, val)
        self.map[key] = node
        self._insert(node)
        if len(self.map) > self.cap:
            lru = self.tail.prev
            self._remove(lru)
            del self.map[lru.key]

cache = LRUCache(capacity=2)
cache.put("concept_oop", 95)
cache.put("concept_apis", 88)
print("Fetch OOP:", cache.get("concept_oop"))

S - SRP
Single Responsibility Principle

ModelEvaluationResult only manages structured outcome representation. BasePredictiveModel manages lifecycle and interface contracts.

O - OCP
Open/Closed Principle

The pipeline is open for extension (you can introduce NeuralNetworkClassifier without touching existing code) and closed for modification.

L - LSP
Liskov Substitution Principle

Any subclass instance (such as CareerPathClassifier or SkillGapEvaluator) can be seamlessly passed anywhere BasePredictiveModel is expected.

I - ISP
Interface Segregation Principle

Interfaces are lean and specialized. Subclasses are only forced to implement predict(), avoiding bloated multi-method base classes.

D - DIP
Dependency Inversion Principle

The pipeline execution engine depends upon the high-level abstraction BasePredictiveModel, rather than concrete low-level implementations.

Critical Software Engineering Pitfalls to Avoid in this Domain

Compare typical anti-patterns against production best practices implemented in this module.

Anti-Pattern: Leaking Encapsulated State 
# BAD: Mutating internal state directly
class Model:
    def __init__(self):
        self.is_ready = False

m = Model()
m.is_ready = True  # Bypasses warm_up() checks!
Best Practice: Controlled Property Getters 
# GOOD: Encapsulated with @property
class Model:
    def __init__(self):
        self._is_ready = False
    @property
    def is_ready(self) -> bool:
        return self._is_ready # Read-only access!
Anti-Pattern: Mutable Default Arguments 
# BAD: Shared mutable list across instances
def train_model(features=[]):
    features.append(1.0)
    return features # Persists across calls!
Best Practice: Sentinel Default Value 
# GOOD: Safe default sentinel assignment
def train_model(features=None):
    if features is None:
        features = []
    features.append(1.0)
    return features
Answer: In Python, inheriting from ABC (Abstract Base Class) and decorating methods with @abstractmethod creates an unyielding interface contract. If a developer creates a subclass and forgets to implement predict(), Python refuses to instantiate the class and immediately raises a TypeError. This guarantees runtime safety in production pipelines.
Answer: Polymorphism literally means "many forms". The pipeline array treats all objects uniformly as BasePredictiveModel. However, at runtime, each instance (e.g., CareerPathClassifier or SkillGapEvaluator) dispatches to its own distinct execution algorithm while returning the exact same structured ModelEvaluationResult format.
Answer: @dataclass(frozen=True) provides static type enforcement, auto-generated immutability, cleaner memory footprint, and compile-time autocomplete in IDEs. Plain dictionaries allow arbitrary mutations, silent spelling errors in key names (leading to runtime KeyError), and zero static type validation.
